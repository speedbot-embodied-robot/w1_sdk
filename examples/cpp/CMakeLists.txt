cmake_minimum_required(VERSION 3.20)
project(w1_docs_cpp_example VERSION 0.1.0 LANGUAGES CXX)

set(is_top_level OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(is_top_level ON)
endif()
option(W1_DOCS_CPP_BUILD_EXAMPLES "Build the command-line example" ${is_top_level})
option(W1_DOCS_CPP_BUILD_TESTS "Build library and command-line tests" ${is_top_level})

add_library(w1_docs_state src/joint_state.cpp)
add_library(w1_docs::state ALIAS w1_docs_state)
target_include_directories(w1_docs_state PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_compile_features(w1_docs_state PUBLIC cxx_std_17)
set_target_properties(w1_docs_state PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(w1_docs_state PRIVATE /W4)
else()
  target_compile_options(w1_docs_state PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(W1_DOCS_CPP_BUILD_EXAMPLES)
  add_executable(joint_state_demo main.cpp)
  target_link_libraries(joint_state_demo PRIVATE w1_docs::state)
  set_target_properties(joint_state_demo PROPERTIES CXX_EXTENSIONS OFF)
endif()

if(W1_DOCS_CPP_BUILD_TESTS)
  enable_testing()
  add_executable(joint_state_test tests/joint_state_test.cpp)
  target_link_libraries(joint_state_test PRIVATE w1_docs::state)
  set_target_properties(joint_state_test PROPERTIES CXX_EXTENSIONS OFF)
  add_test(NAME joint_state_validation COMMAND joint_state_test)
  if(W1_DOCS_CPP_BUILD_EXAMPLES)
    add_test(NAME joint_state_cli COMMAND joint_state_demo)
    set_tests_properties(joint_state_cli PROPERTIES
      PASS_REGULAR_EXPRESSION "demo_joint_2: position=-0.500 rad")
    add_test(NAME joint_state_invalid COMMAND joint_state_demo --invalid)
    set_tests_properties(joint_state_invalid PROPERTIES WILL_FAIL TRUE)
  endif()
endif()
